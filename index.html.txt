let currentCategory = 'all';
let currentFood = null;

// 切换分类
function setCategory(type, btn) {
  currentCategory = type;
  // 更新按钮样式
  document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));
  btn.classList.add('active');
}

// 抽取逻辑
function roll() {
  // 先根据分类过滤（模拟数据库查询）
  let pool = foodData;
  if (currentCategory !== 'all') {
    pool = foodData.filter(food => food.type === currentCategory);
  }

  if (pool.length === 0) {
    alert("这个分类下暂时没有菜品哦！");
    return;
  }

  // 随机抽取
  currentFood = pool[Math.floor(Math.random() * pool.length)];
  
  // 渲染界面
  const resultDiv = document.getElementById('result');
  resultDiv.innerHTML = `
    <div class="food-name">${currentFood.name}</div>
    <div class="food-info">
      <span class="tag">¥${currentFood.price}</span>
      <span class="tag">${currentFood.calorie} 大卡</span>
      <span class="tag" style="color:${currentFood.healthy ? '#27ae60' : '#e74c3c'}">
        ${currentFood.healthy ? '健康' : '高热量'}
      </span>
    </div>
    <div class="food-reason">"${currentFood.reason}"</div>
  `;
  document.getElementById('report-msg').innerText="";
}

// 上报排队情况
function report(type) {
 if(!currentFood){alert("先摇一个食堂再说！");return;}
 document.getElementById('report-msg').innerText="✅ 已上报："+type+"（"+currentFood.name+"）";
}
